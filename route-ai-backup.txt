import {
  NextRequest,
  NextResponse,
} from "next/server";

import {
  getServerSession,
} from "next-auth";

import OpenAI from "openai";

import {
  authOptions,
} from "@/lib/auth";

import {
  connectDB,
} from "@/lib/mongodb";

import Article from "@/models/Article";

import {
  persistAIKnowledgeGrowth,
} from "@/lib/ai/knowledgeGrowth";

export const dynamic =
  "force-dynamic";

type ChatMessage = {
  role:
    | "user"
    | "assistant";

  content: string;
};

type LocalArticle = {
  title: string;
  slug: string;
  summary?: string;
  content?: string;
  category?: string;
};

/* =========================================================
   OPENAI
========================================================= */

function getOpenAIClient() {
  const apiKey =
    process.env
      .OPENAI_API_KEY;

  if (!apiKey) {
    throw new Error(
      "OPENAI_API_KEY is not configured"
    );
  }

  return new OpenAI({
    apiKey,
  });
}

/* =========================================================
   QUESTION KEYWORDS
========================================================= */

function extractKeywords(
  question: string
) {
  const ignoredWords =
    new Set([
      "what",
      "when",
      "where",
      "which",
      "who",
      "why",
      "how",

      "is",
      "are",
      "was",
      "were",

      "the",
      "a",
      "an",

      "and",
      "or",

      "of",
      "to",
      "in",
      "on",
      "for",
      "with",
      "from",

      "explain",
      "calculate",
      "calculation",
      "please",
      "tell",
      "me",
    ]);

  return question
    .toLowerCase()
    .replace(
      /[^a-z0-9\s.&/-]/g,
      " "
    )
    .split(/\s+/)
    .map(
      (
        word
      ) =>
        word.trim()
    )
    .filter(
      (
        word
      ) =>
        word.length >=
          3 &&
        !ignoredWords.has(
          word
        )
    )
    .slice(
      0,
      10
    );
}

/* =========================================================
   ESCAPE REGEX
========================================================= */

function escapeRegex(
  value: string
) {
  return value.replace(
    /[.*+?^${}()|[\]\\]/g,
    "\\$&"
  );
}

/* =========================================================
   PETROHUB SEARCH
========================================================= */

async function searchPetroHub(
  question: string
): Promise<
  LocalArticle[]
> {
  try {
    await connectDB();

    const keywords =
      extractKeywords(
        question
      );

    if (
      keywords.length ===
      0
    ) {
      return [];
    }

    const expressions =
      keywords.map(
        (
          keyword
        ) =>
          new RegExp(
            escapeRegex(
              keyword
            ),
            "i"
          )
      );

    const articles =
      await Article.find({
        status:
          "Published",

        $or: [
          {
            title: {
              $in:
                expressions,
            },
          },

          {
            summary: {
              $in:
                expressions,
            },
          },

          {
            tags: {
              $in:
                expressions,
            },
          },

          {
            category: {
              $in:
                expressions,
            },
          },
        ],
      })
        .select(
          "title slug summary content category"
        )
        .limit(
          4
        )
        .lean();

    return JSON.parse(
      JSON.stringify(
        articles
      )
    ) as LocalArticle[];
  } catch (error) {
    console.error(
      "PetroHub AI local search error:",
      error
    );

    return [];
  }
}

/* =========================================================
   LOCAL CONTEXT
========================================================= */

function buildLocalContext(
  articles:
    LocalArticle[]
) {
  if (
    articles.length ===
    0
  ) {
    return `
No sufficiently relevant PetroHub article was found.

Use trusted web research when necessary.
`;
  }

  return articles
    .map(
      (
        article,
        index
      ) => {
        const content =
          (
            article.content ||
            ""
          ).slice(
            0,
            6000
          );

        return `
PETROHUB SOURCE ${index + 1}

Title:
${article.title}

Category:
${article.category || ""}

Internal PetroHub path:
/articles/${article.slug}

Summary:
${article.summary || ""}

Content:
${content}
`;
      }
    )
    .join(
      "\n\n"
    );
}

/* =========================================================
   REMOVE EXTERNAL LINKS
========================================================= */

function removeExternalUrls(
  text: string
) {
  return text

    /*
      Markdown external links:
      [OSHA](https://...)
      becomes:
      OSHA
    */

    .replace(
      /\[([^\]]+)\]\((https?:\/\/[^)]+)\)/gi,
      "$1"
    )

    /*
      Plain external URLs.
    */

    .replace(
      /https?:\/\/[^\s)\]}]+/gi,
      ""
    )

    .replace(
      /\n{3,}/g,
      "\n\n"
    )

    .trim();
}

/* =========================================================
   POST
========================================================= */

export async function POST(
  request:
    NextRequest
) {
  try {
    /* ===============================================
       AUTH
    =============================================== */

    const session =
      await getServerSession(
        authOptions
      );

    if (
      !session?.user
    ) {
      return NextResponse.json(
        {
          success:
            false,

          message:
            "Please log in to use PetroHub AI.",
        },
        {
          status:
            401,
        }
      );
    }

    /* ===============================================
       BODY
    =============================================== */

    const body =
      await request.json();

    const messages:
      ChatMessage[] =
      Array.isArray(
        body.messages
      )
        ? body.messages
        : [];

    const validMessages =
      messages
        .filter(
          (
            message
          ) =>
            message &&
            (
              message.role ===
                "user" ||
              message.role ===
                "assistant"
            ) &&
            typeof message.content ===
              "string" &&
            message.content.trim()
        )
        .slice(
          -12
        );

    if (
      validMessages.length ===
      0
    ) {
      return NextResponse.json(
        {
          success:
            false,

          message:
            "Please enter a question.",
        },
        {
          status:
            400,
        }
      );
    }

    const latestUserMessage =
      [...validMessages]
        .reverse()
        .find(
          (
            message
          ) =>
            message.role ===
            "user"
        );

    if (
      !latestUserMessage
    ) {
      return NextResponse.json(
        {
          success:
            false,

          message:
            "No user question was provided.",
        },
        {
          status:
            400,
        }
      );
    }

    /* ===============================================
       PETROHUB DB FIRST
    =============================================== */

    const localArticles =
      await searchPetroHub(
        latestUserMessage.content
      );

    const localContext =
      buildLocalContext(
        localArticles
      );

    /* ===============================================
       OPENAI
    =============================================== */

    const openai =
      getOpenAIClient();

    const model =
      process.env
        .OPENAI_MODEL ||
      "gpt-5.6-luna";

    const response =
      await openai.responses.create(
        {
          model,

          store:
            false,

          max_output_tokens:
            1800,

          tools: [
            {
              type:
                "web_search",
            },
          ],

          instructions: `
You are PetroHub AI, the engineering knowledge assistant inside PetroHub.

==================================================
PRIMARY TECHNICAL DOMAIN
==================================================

Your primary expertise is:

- Petroleum Engineering
- Drilling Engineering
- Well Engineering
- Reservoir Engineering
- Production Engineering
- Oil & Gas Processing
- Refining
- Petrochemical Engineering
- Process Engineering
- Fluid Mechanics
- Thermodynamics
- Heat Transfer
- Pipeline Engineering
- Petroleum Fluid Properties
- Petroleum-related HSE
- Process Safety

==================================================
KNOWLEDGE STRATEGY
==================================================

1. PetroHub's own published knowledge is supplied below.

2. Use PetroHub knowledge first when it adequately answers the question.

3. If PetroHub knowledge is missing, incomplete, outdated, or insufficient, use web search.

4. When web research is used, prefer authoritative technical sources:
   - government agencies,
   - recognized engineering organizations,
   - standards bodies,
   - universities,
   - established technical institutions,
   - original manufacturer documentation.

5. Give the complete useful answer inside PetroHub.

6. Never require the user to visit another website to obtain the answer.

7. Do not expose external URLs in the final answer.

8. Never pretend external research originated from PetroHub.

==================================================
OUTPUT FORMAT
==================================================

Your response is rendered using Markdown + KaTeX.

HEADINGS:

Use:

## Main heading

### Subheading

Never place Markdown headings inside mathematical equations.

--------------------------------------------------
DISPLAY EQUATIONS
--------------------------------------------------

Every standalone equation MUST use:

$$
P = 0.052 \\times MW \\times TVD
$$

Rules:

- Always use TWO dollar signs for display equations.
- Never use a single dollar sign on its own line.
- Never place Markdown headings, bullets, backticks, or prose inside math blocks.
- Every opening $$ must have a closing $$.

--------------------------------------------------
INLINE EQUATIONS
--------------------------------------------------

Use:

$SG = 0.85$

Do not put mathematical variables in backticks.

--------------------------------------------------
VARIABLES
--------------------------------------------------

After formulas define variables using bullets.

Example:

- $P$ = hydrostatic pressure, psi
- $MW$ = mud weight, ppg
- $TVD$ = true vertical depth, ft

--------------------------------------------------
CALCULATIONS
--------------------------------------------------

For calculations use:

### Given

### Formula

### Substitution

### Result

Example:

### Given

- $MW = 10\\ \\text{ppg}$
- $TVD = 8000\\ \\text{ft}$

### Formula

$$
P = 0.052 \\times MW \\times TVD
$$

### Substitution

$$
P = 0.052 \\times 10 \\times 8000
$$

### Result

$$
\\boxed{P = 4160\\ \\text{psi}}
$$

--------------------------------------------------
TABLES
--------------------------------------------------

Only create a Markdown table when genuinely useful.

Every table must contain:
- header row,
- separator row,
- equal number of columns.

For simple classifications prefer bullet points.

--------------------------------------------------
NO DUPLICATION
--------------------------------------------------

Never repeat the same:
- definition,
- formula,
- example,
- section,
- conclusion.

Before sending the answer silently check for duplicated sections.

==================================================
FORMULA ACCURACY
==================================================

Never invent:
- formulas,
- coefficients,
- conversion factors,
- physical properties,
- standard limits,
- design limits.

State assumptions and applicability when relevant.

==================================================
SAFETY
==================================================

For safety-critical, pressure-containing, relief-system, well-control, structural, hazardous-area, process-safety, or regulatory matters:

- give useful technical guidance,
- state important assumptions,
- distinguish educational/general calculations from project-specific engineering design,
- never fabricate requirements.

==================================================
LANGUAGE
==================================================

The user may use:
- English,
- Tamil,
- Tamil written using English letters.

Understand all three.

Preserve standard engineering notation.

==================================================
PETROHUB KNOWLEDGE
==================================================

${localContext}
`,

          input:
            validMessages.map(
              (
                message
              ) => ({
                role:
                  message.role,

                content:
                  message.content,
              })
            ),
        }
      );

    /* ===============================================
       ANSWER
    =============================================== */

    const rawAnswer =
      response.output_text?.trim();

    if (
      !rawAnswer
    ) {
      throw new Error(
        "The AI returned an empty response"
      );
    }

    const answer =
      removeExternalUrls(
        rawAnswer
      );

    /* ===============================================
       WEB SEARCH DETECTION
    =============================================== */

    const usedWebSearch =
      response.output.some(
        (
          item:
            any
        ) =>
          item?.type ===
          "web_search_call"
      );

    /* ===============================================
       MODE
    =============================================== */

    let mode:
      | "petrohub"
      | "web"
      | "hybrid" =
      "petrohub";

    if (
      localArticles.length ===
        0 &&
      usedWebSearch
    ) {
      mode =
        "web";
    } else if (
      localArticles.length >
        0 &&
      usedWebSearch
    ) {
      mode =
        "hybrid";
    }

    /* ===============================================
       KNOWLEDGE GROWTH
    =============================================== */

    /* ===============================================
   USER SESSION
=============================================== */

const sessionUser =
  session.user as {
    id?: string;
    email?: string | null;
  };

/* ===============================================
   KNOWLEDGE GROWTH DEBUG

   This tells us exactly whether the answer
   came from PetroHub, web research or hybrid.
=============================================== */

console.log(
  "PETROHUB AI KNOWLEDGE DEBUG",
  {
    question:
      latestUserMessage.content,

    mode,

    usedWebSearch,

    localArticleCount:
      localArticles.length,

    hadLocalSources:
      localArticles.length > 0,
  }
);

/* ===============================================
   KNOWLEDGE GROWTH

   This logs the question and,
   when web research is involved,
   creates/links a Draft article.
=============================================== */

await persistAIKnowledgeGrowth({
  userId:
    sessionUser.id ||
    sessionUser.email ||
    "unknown-user",

  email:
    sessionUser.email ||
    "",

  question:
    latestUserMessage.content,

  answer,

  mode,

  hadLocalSources:
    localArticles.length > 0,

  localSourceCount:
    localArticles.length,

  usedWebSearch,
});

/* ===============================================
   RESPONSE
=============================================== */

return NextResponse.json({
  success: true,

  answer,

  mode,

  usedWebSearch,

  localSources:
    localArticles.map(
      (article) => ({
        title:
          article.title,

        slug:
          article.slug,
      })
    ),
}
